# nginx: one server block for each service. ./deltactl access --apply puts each one in /etc/nginx/sites-available/ and links it in sites-enabled/; by hand: nginx -t && systemctl reload nginx
# (this text is only written under /etc/nginx when ./deltactl access --apply is run)

# s0
server {
    listen 80;
    server_name backend-s0.ejemplo.pe;
    return 301 https://$host$request_uri;
}
server {
    listen 443 ssl;
    server_name backend-s0.ejemplo.pe;
    # the certificate: certbot --nginx -d backend-s0.ejemplo.pe
    ssl_certificate /etc/letsencrypt/live/backend-s0.ejemplo.pe/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/backend-s0.ejemplo.pe/privkey.pem;
    location / {
        proxy_pass http://127.0.0.1:4100;

        # Requerido para WebSockets
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Reenvío de cabeceras importantes
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        # the protocol the person used: what the server in front says when it ends HTTPS, else this one
        set $deltactl_proto $scheme;
        if ($http_x_forwarded_proto != "") { set $deltactl_proto $http_x_forwarded_proto; }
        proxy_set_header X-Forwarded-Proto $deltactl_proto;

        # Evitar CACHE
        proxy_cache_bypass $http_upgrade;
    }
}
